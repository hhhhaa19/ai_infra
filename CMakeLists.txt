cmake_minimum_required(VERSION 3.18)
# Default for our A100 experiments; override for other GPU architectures.
set(CMAKE_CUDA_ARCHITECTURES 80 CACHE STRING "CUDA architectures")
project(AiInfraLearning LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)

set(TENSORRT_ROOT "$ENV{TENSORRT_ROOT}" CACHE PATH "TensorRT SDK root")
find_path(TENSORRT_INCLUDE_DIR NvInfer.h HINTS "${TENSORRT_ROOT}/include")
find_library(TENSORRT_LIBRARY nvinfer HINTS "${TENSORRT_ROOT}/lib" "${TENSORRT_ROOT}/lib64")
find_library(TENSORRT_ONNX_LIBRARY nvonnxparser HINTS "${TENSORRT_ROOT}/lib" "${TENSORRT_ROOT}/lib64")
if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY OR NOT TENSORRT_ONNX_LIBRARY)
    message(FATAL_ERROR "Set TENSORRT_ROOT to a TensorRT C++ SDK")
endif()
foreach(target build_engine infer infer_async infer_pageable)
    add_executable(${target} src/tensorrt_intro/${target}.cpp)
    target_include_directories(${target} PRIVATE "${TENSORRT_INCLUDE_DIR}")
    target_link_libraries(${target} PRIVATE CUDA::cudart "${TENSORRT_LIBRARY}" "${TENSORRT_ONNX_LIBRARY}")
endforeach()
add_executable(main src/basics/main.cpp)
target_link_libraries(main PRIVATE CUDA::cudart)

function(add_cuda_example target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE CUDA::cudart)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O2> $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endfunction()
add_cuda_example(vector_add src/basics/vector_add.cu)
foreach(target reduce_sum warp_reduce reduction_bench)
    add_cuda_example(${target} src/reduction/${target}.cu)
endforeach()
# RMSNorm 独立 CUDA 测试：可执行文件名保留旧名（报告里用的是这些名字），源码在 src/rmsnorm/cuda_bench/
add_cuda_example(rmsnorm_n_bench src/rmsnorm/cuda_bench/n_bench.cu)
add_cuda_example(rmsnorm_cub src/rmsnorm/cuda_bench/cub_example.cu)
add_cuda_example(rmsnorm_cub_bench src/rmsnorm/cuda_bench/cub_bench.cu)
add_cuda_example(rmsnorm_profile src/rmsnorm/cuda_bench/profile.cu)

# TensorRT plugin：ONNX 里的 RmsNorm 节点用我们的 kernel（trtexec --plugins=build-orin/librmsnorm_trt_plugin.so）
add_library(rmsnorm_trt_plugin SHARED src/rmsnorm/tensorrt/plugin.cu)
target_include_directories(rmsnorm_trt_plugin PRIVATE "${TENSORRT_INCLUDE_DIR}")
target_link_libraries(rmsnorm_trt_plugin PRIVATE CUDA::cudart "${TENSORRT_LIBRARY}")
target_compile_options(rmsnorm_trt_plugin PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O2> $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
